  <script>
    const states = {
      initial: "Lorem ipsum dolor sit amet, consectetur adipiscing elit. Curabitur libero massa semper pretium. Donec congue justo sed placerat consequat, velit erat volutpat lorem eget posuere risus justo eget arcu.",
      one: "Phasellus aliquam turpis vel erat malesuada, ac porta metus condimentum. Integer tempus lacus vitae nibh consequat, nec ullamcorper ligula faucibus. Suspendisse potenti. Morbi gravida turpis vitae felis accumsan lobortis.",
      two: "Integer bibendum magna sed nisl maximus, vitae dignissim augue facilisis. Nam pulvinar luctus magna, quis interdum mauris volutpat eget. Curabitur consequat tellus eget risus posuere, sed volutpat mi ultrices."
    };

    const wordsEl = document.getElementById("words");
    const ghostCopy = document.getElementById("ghostCopy");
    const textStage = document.getElementById("textStage");
    const buttons = [...document.querySelectorAll(".choice")];
    const reducedMotion = window.matchMedia("(prefers-reduced-motion: reduce)");

    let currentKey = "initial";
    let isAnimating = false;
    let timers = [];

    function renderText(layer, text) {
      const fragment = document.createDocumentFragment();
      text.trim().split(/\s+/).forEach((word) => {
        const span = document.createElement("span");
        span.className = "word";
        span.textContent = word;
        fragment.append(span);
      });
      layer.replaceChildren(fragment);
    }

    function updateButtons(activeKey) {
      buttons.forEach((button) => {
        const active = button.dataset.state === activeKey;
        button.classList.toggle("active", active);
        button.setAttribute("aria-pressed", String(active));
      });
    }

    function clearTimers() {
      timers.forEach(window.clearTimeout);
      timers = [];
    }

    function later(callback, delay) {
      timers.push(window.setTimeout(callback, delay));
    }

    function switchText(nextKey) {
      if (isAnimating || nextKey === currentKey || !states[nextKey]) return;

      if (reducedMotion.matches) {
        renderText(wordsEl, states[nextKey]);
        updateButtons(nextKey);
        currentKey = nextKey;
        return;
      }

      isAnimating = true;
      clearTimers();
      buttons.forEach((button) => button.disabled = true);

      renderText(ghostCopy, states[currentKey]);
      wordsEl.classList.remove("is-arriving");
      ghostCopy.classList.remove("old-echo", "new-echo");
      textStage.classList.remove("is-burst");
      void textStage.offsetWidth;

      textStage.classList.add("is-burst");
      wordsEl.classList.add("is-leaving");
      ghostCopy.classList.add("old-echo");

      later(() => {
        renderText(wordsEl, states[nextKey]);
        renderText(ghostCopy, states[nextKey]);
        updateButtons(nextKey);
        currentKey = nextKey;

        wordsEl.classList.remove("is-leaving", "is-arriving");
        ghostCopy.classList.remove("old-echo", "new-echo");
        void wordsEl.offsetWidth;
        wordsEl.classList.add("is-arriving");
        ghostCopy.classList.add("new-echo");
      }, 360);

      later(() => {
        wordsEl.classList.remove("is-arriving");
        ghostCopy.classList.remove("old-echo", "new-echo");
        textStage.classList.remove("is-burst");
        buttons.forEach((button) => button.disabled = false);
        isAnimating = false;
      }, 2000);
    }

    renderText(wordsEl, states.initial);
    updateButtons(currentKey);

    buttons.forEach((button) => {
      button.addEventListener("click", () => switchText(button.dataset.state));
    });
  </script>
